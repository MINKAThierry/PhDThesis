\section*{ACRONYMES}
\subsection*{A-C}
\begin{sloppypar}

\item[ENSP] \'Ecole Nationale Sup\'erieure Polytechnique (\textit{National Advanced School of Engineering})
\item[ESIGN] Electronic Signatures in Global and National Commerce Act, loi am\'ericaine de 2000

\item[SOTP-High] SOTP au-del\`a du seuil $H^\sharp$
\item[SOTP-Low] SOTP sous le seuil $H^\sharp$, lorsque l'institution a besoin du contexte exact

\section*{SYMBOLES MATH\'EMATIQUES ET NOTATIONS}

\subsection*{Param\`etres institutionnels}

\begin{description}[style=nextline,leftmargin=1.6cm,font=\normalfont\bfseries]
\item[$\Delta$] Capacit\'e d'extraction institutionnelle (bits), $I(V;\Sigma|M,C) \leq \Delta$~; quelques dizaines de bits en calibration illustrative (D\'efinition~\ref{def:delta_extraction})
\item[$\Delta^*$] Capacit\'e propre \`a l'institution, plus petite valeur de $\Delta$ qui satisfait l'Hypoth\`ese~\ref{ass:extraction_bornee} (Remarque~\ref{rem:delta_capacite})
\item[$\Delta_q$] Capacit\'e d'extraction institutionnelle adapt\'ee au quantum, $\Delta_q = \Delta + \delta_{\mathrm{enh}} - \delta_{\mathrm{degr}}$, o\`u $\delta_{\mathrm{enh}}$ est l'am\'elioration quantique et $\delta_{\mathrm{degr}}$ la p\'enalit\'e d'adaptation (D\'efinition~\ref{def:decomposition_delta_q})
\item[$\alpha$] Granularit\'e des verdicts institutionnels, $\log_2|\mathcal{V}_\lambda| \leq \alpha H(C)$
\item[$\alpha_q$, $\beta_q$, $\zeta_q$, $\theta_q$] Param\`etres du cadre quantique, soit la granularit\'e $\alpha_q$, les constantes de r\'egularit\'e $\beta_q$ et $\zeta_q$ et la non-d\'eg\'en\'erescence du verdict $\theta_q \in (0,1]$ (Hypoth\`eses~\ref{ass:verdict_quantique} et~\ref{ass:regularite_quantique})
\item[$\beta$] Constante de r\'egularit\'e du protocole, $\mathrm{KL}(P_{\Sigma|m_0,c}\|P_{\Sigma|m_1,c}) \leq \beta \cdot I(M;\Sigma|C)$~; elle garantit la confidentialit\'e (Lemme~\ref{lem:confidentiality_guarantee})
\item[$\gamma$] Minorant de l'entropie des verdicts, $\gamma = \alpha H(C) - 1 \geq 1$ (Hypoth\`ese~\ref{ass:verdicts_efficace})~; $\gamma_q = \theta_q\alpha_q H(C)$ dans le cadre quantique
\item[$\eta_q$] Coefficient de perte d'interpr\'etabilit\'e quantique, plus petit r\'eel tel que $I(V;\Lambda(\Sigma)|C) \geq I(V;\Sigma|C) - \eta_q H(C)$ (D\'efinition~\ref{def:perte_interp_quantique})~; $\eta_q = 0$ pour une preuve qui est une cha\^{\i}ne de bits
\item[$\kappa$] Constante du r\'egime de pr\'eservation de la confidentialit\'e, $I(M;\Sigma|C) \leq \kappa/\lambda$ (Hypoth\`ese~\ref{ass:regime})
\item[$H(C)$] Entropie du contexte institutionnel~; not\'ee aussi $H(\mathcal{C})$, les deux \'ecritures d\'esignant la m\^eme quantit\'e
\end{description}

\subsection*{Param\`etres cryptographiques}

\begin{description}[style=nextline,leftmargin=1.5cm,font=\normalfont\bfseries]
\item[$\lambda$] Param\`etre de s\'ecurit\'e cryptographique, typiquement $\lambda \in \{128, 192, 256\}$
\item[$\mathsf{negl}(\lambda)$] Fonction n\'egligeable en $\lambda$, $\forall c>0, \exists \lambda_0, \forall \lambda > \lambda_0: |f(\lambda)| < \lambda^{-c}$
\item[$c_0$] Constante de Pinsker, $c_0 = 1/(2\ln 2) \approx 0{,}721$, avec la divergence en bits~; la constante serr\'ee est $(\ln 2)/2 \approx 0{,}347$ (Remarque~\ref{rem:constante_c0})
\item[$H^\sharp$] Seuil de la forme raffin\'ee, $H^\sharp = (\Delta + 1 + \kappa/\lambda)/\alpha$~; au-del\`a de ce seuil, la capacit\'e institutionnelle est contraignante (Remarque~\ref{rem:activite_delta})
\item[$H^\sharp_q$] Seuil quantique, $H^\sharp_q = (\Delta_q + \kappa/\lambda)/(\theta_q\alpha_q)$ (Remarque~\ref{rem:forme_raffinee_pq})
\end{description}

\subsection*{M\'etriques CRO}

\begin{description}[style=nextline,leftmargin=1.7cm,font=\normalfont\bfseries]
\item[$\mathsf{Conf}(\mathcal{P})$] Confidentialit\'e du protocole $\mathcal{P}$, $\mathsf{Conf} = 1 - \max_{m_0,m_1,c}\mathsf{TV}(P_{\Sigma|m_0,c}, P_{\Sigma|m_1,c})$ (D\'efinition~\ref{def:confidentialite_clo})
\item[$\Relrec(\mathcal{P})$] Fiabilit\'e de reconstruction, $\Relrec = 1 - \min_{\hat M}\Pr[\hat{M}(\Sigma,C) \neq M]$ (contr\^ol\'ee par Fano)~; not\'ee aussi $\mathsf{Rel}$ (D\'efinition~\ref{def:fiabilite_clo})
\item[$\Relver(\mathcal{P})$] Fiabilit\'e de v\'erification, compl\'etude et solidit\'e du v\'erificateur (Remarque~\ref{rem:rel_ver})
\item[$\mathsf{Opp}(\mathcal{P})$] Opposabilit\'e du protocole $\mathcal{P}$, $\mathsf{Opp} = I(V;\Sigma|C)/H(V|C)$ (D\'efinition~\ref{def:opposabilite_clo})
\item[$\mathsf{Conf}^q$] Confidentialit\'e quantique, $1 - \max_{m_0,m_1,c} \tfrac12\|\rho_\Sigma^{m_0,c}-\rho_\Sigma^{m_1,c}\|_1$, au sens statistique (D\'efinition~\ref{def:conf_quantique_clo})
\item[$\mathsf{Rel}^q$] Fiabilit\'e quantique de reconstruction, not\'ee aussi $\Relrec^q$, recouvrement optimal par une POVM
\item[$\mathsf{Opp}^q$] Opposabilit\'e quantique, $I(V;\Sigma|C)/H(V|C)$ pour un verdict admissible rendu sur l'issue de la mesure (D\'efinition~\ref{def:opposabilite_quantique})
\item[$\mathsf{Opp}^{\Lambda}$] Opposabilit\'e apr\`es passage de la preuve dans le canal $\Lambda$, $I(V;\Lambda(\Sigma)|C)/H(V|C)$
\end{description}

\subsection*{Variables al\'eatoires et espaces}

\begin{description}[style=nextline,leftmargin=1cm,font=\normalfont\bfseries]
\item[$M$] Variable al\'eatoire message, espace $\mathcal{M}_\lambda$ avec $|\mathcal{M}_\lambda| = 2^\lambda$
\item[$\Sigma$] Variable al\'eatoire preuve cryptographique, espace $\Sigma_\lambda$
\item[$C$] Variable al\'eatoire contexte institutionnel, espace $\mathcal{C}_\lambda$ avec distribution $P_C$
\item[$V$] Variable al\'eatoire verdict institutionnel, $V = \mathsf{Interpret}(\Sigma, C, K)$ avec $K$ les ressources de l'institution
\item[$K$] Ressources propres de l'institution, avec $K \perp M \mid (\Sigma, C)$ (D\'efinition~\ref{def:interp_admissible})
\item[$\mathcal{V}_\lambda$] Espace des verdicts, $|\mathcal{V}_\lambda| \leq 2^{\alpha H(C)}$
\item[$\rho$] \'Etat quantique (matrice de densit\'e)
\item[$\Lambda$] Canal quantique (application CPTP)
\end{description}

\subsection*{Mesures d'information}

\begin{description}[style=nextline,leftmargin=2cm,font=\normalfont\bfseries]
\item[$H(X)$] Entropie de Shannon, $H(X) = -\sum_x P_X(x)\log_2 P_X(x)$
\item[$H(X|Y)$] Entropie conditionnelle, $H(X|Y) = H(X,Y) - H(Y)$
\item[$h(p)$] Entropie binaire, $h(p) = -p\log_2 p - (1-p)\log_2(1-p)$~; not\'ee $h_2$ au Lemme~\ref{lem:fano_quantique}
\item[$I(X;Y)$] Information mutuelle, $I(X;Y) = H(X) - H(X|Y) = H(Y) - H(Y|X)$
\item[$I(X;Y|Z)$] Information mutuelle conditionnelle, $I(X;Y|Z) = H(X|Z) - H(X|Y,Z)$
\item[$D_{\mathrm{KL}}(P\|Q)$] Divergence de Kullback-Leibler, $D_{\mathrm{KL}}(P\|Q) = \sum_x P(x)\log_2\frac{P(x)}{Q(x)}$
\item[$\mathsf{TV}(P,Q)$] Distance de variation totale, $\mathsf{TV}(P,Q) = \frac{1}{2}\sum_x|P(x)-Q(x)|$
\item[$S(\rho)$] Entropie de von Neumann, $S(\rho) = -\mathrm{Tr}(\rho\log\rho)$
\item[$\chi$] Information de Holevo, $\chi = S(\bar{\rho}) - \sum_x P_X(x)S(\rho_x)$
\item[$I_{\mathrm{acc}}$] Information accessible, supr\'emum sur toutes les POVMs
\end{description}

\subsection*{Notions quantiques}

\begin{description}[style=nextline,leftmargin=1.6cm,font=\normalfont\bfseries]
\item[$\|\rho - \sigma\|_1$] Distance de trace quantique, $\mathrm{Tr}\sqrt{(\rho-\sigma)^\dagger(\rho-\sigma)}$
\item[$F(\rho,\sigma)$] Fid\'elit\'e quantique, $F(\rho,\sigma) = \|\sqrt{\rho}\sqrt{\sigma}\|_1$
\item[$\|\Phi\|_\diamond$] Norme diamant, mesure de distance entre canaux quantiques (D\'efinition~\ref{def:norme_diamant})
\item[$\Lambda_{\mathrm{dep}}$] Canal d\'epolarisant, $\Lambda_{\mathrm{dep}}(\rho) = (1-p)\rho + \frac{p}{d}I_d$
\item[$\Lambda_{\mathrm{AD}}$] Canal d'amortissement d'amplitude, de param\`etre $g$ (D\'efinition~\ref{def:canal_amortissement})
\item[$\Lambda_{\mathrm{deph}}$] Canal de d\'ephasage, $\Lambda_{\mathrm{deph}}(\rho) = (1-p)\rho + pZ\rho Z$ (D\'efinition~\ref{def:canal_dephasage})
\end{description}

\subsection*{Formalisation Lean4}

\begin{description}[style=nextline,leftmargin=4.3cm,font=\normalfont\bfseries]
\item[\texttt{ProtocolFamily}] Structure Lean4 repr\'esentant une famille de protocoles index\'ee par $\lambda$
\item[\texttt{ProtocolInstance}] Structure Lean4 pour une instance \`a $\lambda$ fix\'e
\item[\texttt{InstitutionalParams}] Structure Lean4 pour les param\`etres $(\Delta, \alpha, \beta, H)$
\item[\texttt{Negligible}] Pr\'edicat Lean4 pour les fonctions n\'egligeables
\item[\texttt{atTop}] Filtre Lean4 pour les comportements asymptotiques
%\item[Quasi-optimalit\'e] Hypoth\`ese $I(M;\Sigma|C) \leq 1$ pour $\lambda$ suffisamment grand
\end{description}

\section*{CONCEPTS CL\'ES DE LA CRYPTOGRAPHIE INSTITUTIONNELLE}

\subsection*{Paradigme architectural}

\begin{description}[style=nextline,leftmargin=2.5cm,font=\normalfont\bfseries]
\item[Couche Gold] Couche de confidentialit\'e, masque jetable (tronqu\'e ou pseudo-al\'eatoire) ou preuve \`a divulgation nulle
\item[Couche Iron] Couche de fiabilit\'e de v\'erification, engagements liants (\textit{binding})
\item[Couche Clay] Couche de l'opposabilit\'e, qui pr\'esente le contexte institutionnel \`a l'interpr\`ete. Elle accro\^it $\Opp$ lorsqu'elle transmet un contexte d'esp\`ece que l'institution ne d\'etient pas d\'ej\`a. Sinon, elle agit comme un format de pr\'esentation (Remarque~\ref{rem:statut_sotp})
\end{description}

\subsection*{Th\'eor\`emes fondamentaux}

%\subsection*{Th\'eor\`emes fondamentaux}

\begin{description}[style=nextline,leftmargin=6cm,font=\normalfont\bfseries]
\item[Borne aigu\"e] $\Relrec \leq \frac{2\kappa}{\lambda(\lambda-3)} + 2^{-\lambda/2}$ sous le r\'egime $I(M;\Sigma|C) \leq \kappa/\lambda$, d'ordre $\lambda^{-2}$ exact (Proposition~\ref{prop:borne_aigue})
\item[Fuchs-van de Graaf] $1 - \frac{1}{2}\|\rho-\sigma\|_1 \leq F(\rho,\sigma) \leq \sqrt{1 - \frac{1}{4}\|\rho-\sigma\|_1^2}$
\item[Goulot d'\'etranglement contextuel] $I(V;\Sigma|C) \leq \min\{H(\Sigma|C), H(V|C)\} \leq \alpha H(\mathcal{C})$ (Th\'eor\`eme~\ref{thm:goulot_contexte})
\item[In\'egalit\'e de Fano] $H(X|\hat{X}) \leq h(p_e) + p_e\log_2(|\mathcal{X}|-1)$ (Lemme~\ref{lem:fano})
\item[In\'egalit\'e de Pinsker] $\mathsf{TV}(P,Q)^2 \leq c_0 \cdot D_{\mathrm{KL}}(P\|Q)$ (Lemme~\ref{lem:pinsker})
\item[Lemme de mesure douce] (\textit{gentle measurement}) $\|\rho - \rho'\|_1 \leq 2\sqrt{\delta}$ pour une issue de probabilit\'e $\geq 1-\delta$ (Lemme~\ref{lem:mesure_douce_etat_art})
\item[Trilemme CRO] $\mathsf{Conf} \cdot \Relrec \cdot \mathsf{Opp} \leq \frac{1}{\gamma}\bigl(\frac{\Delta}{\lambda} + \frac{\kappa(\Delta+1)}{\lambda^2} + \frac{\kappa^2}{\lambda^3}\bigr)$, $\gamma = \alpha H(\mathcal{C}) - 1$~; forme raffin\'ee $\frac{1+\kappa/\lambda}{\lambda}\min\{1, \frac{\kappa/\lambda+\Delta}{\gamma}\}$ (Th\'eor\`eme~\ref{thm:trilemme_cro_conditionnel})
\item[Trilemme CRO post-quantique] (PQ-CRO) $\mathsf{Conf}^q \cdot \Relrec^q \cdot \mathsf{Opp}^q \leq \frac{1}{\gamma_q}\bigl(\frac{\Delta_q}{\lambda} + \frac{\kappa(\Delta_q+1)}{\lambda^2} + \frac{\kappa^2}{\lambda^3}\bigr)$~; majoration affaiblie $(\Delta_q + \eta_q H(C))/\lambda + O(\lambda^{-2}) + \mathsf{negl}(\lambda)$ (Th\'eor\`eme~\ref{thm:trilemme_pq})
\end{description}

\subsection*{R\'egimes op\'erationnels}

\begin{description}[style=nextline,leftmargin=4cm,font=\normalfont\bfseries]
\item[R\'egime \`a contexte rare] $H(C) \leq H^\sharp$ (ou $H^\sharp_q$), l'in\'egalit\'e de Fano limite le produit (Remarque~\ref{rem:activite_delta})
\item[R\'egime \`a contexte riche] $H(C) > H^\sharp$ (ou $H^\sharp_q$), la capacit\'e $\Delta$ (ou $\Delta_q$) limite le produit (Remarque~\ref{rem:activite_delta})
\item[R\'egime de basse capacit\'e] $\Delta < 0{,}1\lambda$, m\'ecanismes transparents recommand\'es (D\'efinition~\ref{def:regime_low}~; \texttt{LowCapacity} en Lean)
\item[R\'egime de capacit\'e moyenne] $0{,}1\lambda \leq \Delta \leq 0{,}5\lambda$, compromis CRO \'equilibr\'es possibles (D\'efinition~\ref{def:regime_mid}~; \texttt{MidCapacity} en Lean)
\item[R\'egime de haute capacit\'e] $\Delta > 0{,}5\lambda$, preuves cryptographiques avanc\'ees soutenables (D\'efinition~\ref{def:regime_high}~; \texttt{HighCapacity} en Lean)
\end{description}

\subsection*{Syst\`emes analys\'es}

\begin{description}[style=nextline,leftmargin=2.5cm,font=\normalfont\bfseries]
\item[Dilithium] Signature PQC fond\'ee sur les r\'eseaux, standardis\'ee sous le nom ML-DSA (FIPS 204)
\item[ECDSA] Signatures num\'eriques sur courbes elliptiques, $\mathsf{Conf} \approx 0{,}31$, $\Relver \approx 0{,}98$, $\mathsf{Opp} \approx 0{,}90$ (calibration)
\item[Falcon] Signature PQC fond\'ee sur les r\'eseaux, \`a l'origine du standard FIPS 206 (FN-DSA) en cours de finalisation
\item[Helios] Protocole de vote \`a v\'erification de bout en bout, $\mathsf{Conf} \approx 0{,}92$, $\Relver \approx 0{,}95$, $\mathsf{Opp} \approx 0{,}50$ (calibration)
\item[Kyber] KEM PQC fond\'e sur les r\'eseaux, standardis\'e sous le nom ML-KEM (FIPS 203)~; hors du champ du Trilemme
\item[SPHINCS+] Signature PQC fond\'ee sur les fonctions de hachage, standardis\'ee sous le nom SLH-DSA (FIPS 205)
\item[zk-SNARKs] Preuves succinctes de conformit\'e, $\mathsf{Conf} \approx 0{,}89$, $\Relver \approx 0{,}96$, $\mathsf{Opp} \approx 0{,}67$ (calibration)
\end{description}
